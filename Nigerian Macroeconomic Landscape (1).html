import pandas as pd
import os

path = r'C:\Users\user\Downloads' 

def load_cbn_file(target_name, skip_rows=0):
    for file in os.listdir(path):
        if target_name.lower() in file.lower():
            file_path = os.path.join(path, file)
            print(f"Attempting to load: {file}")
            
            # Check the file extension
            if file.endswith('.xlsx') or file.endswith('.xls'):
                return pd.read_excel(file_path, skiprows=skip_rows)
            else:
                return pd.read_csv(file_path, encoding='latin1', skiprows=skip_rows, on_bad_lines='skip')
    return None

import pandas as pd
import matplotlib.pyplot as plt

# 1. LOAD DATA (Using your original loader logic)
df = load_cbn_file('International_Payments')
df1 = load_cbn_file('Currency_in_Circulation')
df2 = load_cbn_file('DOMESTIC', skip_rows=1)

# 2. CLEANING & PREPARATION
# Convert dates to datetime objects
df['payDate'] = pd.to_datetime(df['payDate'], dayfirst=True)
df1['volDate'] = pd.to_datetime(df1['volDate'], dayfirst=True)

# Ensure numeric types for International Payments
for col in ['loc', 'remit', 'debt', 'total']:
    df[col] = pd.to_numeric(df[col], errors='coerce')

# FIX: Rename 'amount' to 'currency_amount' for df1
if 'amount' in df1.columns:
    df1 = df1.rename(columns={'amount': 'currency_amount'})
df1['currency_amount'] = pd.to_numeric(df1['currency_amount'], errors='coerce')

# Sort chronologically for growth calculations
df1 = df1.sort_values('volDate')
df = df.sort_values('payDate')

# 3. CALCULATE MACRO TRENDS
# Monthly growth rate of cash
df1['monthly_growth_%'] = df1['currency_amount'].pct_change() * 100

# Aggregate International Payments by Month
df_monthly_int = df.set_index('payDate').resample('ME')['total'].sum().reset_index()

# Merge local currency data with international outflow data
df_macro = pd.merge(df_monthly_int, df1, left_on='payDate', right_on='volDate', how='inner')

# Calculate Outflow Intensity (Ratio of Int. Payments vs Local Cash)
df_macro['outflow_intensity'] = df_macro['total'] / df_macro['currency_amount']

# 4. FINAL VISUALIZATION
fig, ax1 = plt.subplots(figsize=(14, 7))

# Plot 1: Cash Volume (Bar Chart)
ax1.bar(df1['volDate'], df1['currency_amount'], color='lightgrey', alpha=0.5, label='Physical Cash Volume')
ax1.set_ylabel('Total Currency (Millions ₦)', color='grey')

# Plot 2: Growth Rate (Line Chart on Secondary Axis)
ax2 = ax1.twinx()
ax2.plot(df1['volDate'], df1['monthly_growth_%'], color='red', marker='o', markersize=4, label='Monthly Growth %', linewidth=1)
ax2.axhline(0, color='black', linestyle='--')
ax2.set_ylabel('Monthly Growth Rate (%)', color='red')

plt.title('Nigeria: Currency Supply Volatility & Recovery (2002-2024)', fontsize=16)
fig.tight_layout()
plt.show()

print("Final Analysis Complete! Here are your latest trends:")
display(df_macro[['volDate', 'currency_amount', 'monthly_growth_%', 'outflow_intensity']].tail())

Executive Summary: Nigerian Macroeconomic Landscape (2002–2024)

This analysis investigated the relationship between Nigeria's physical currency supply and its international financial obligations.

The "Naira Redesign" Impact: The data provides statistical evidence of the 2023 cash crunch, showing a record-breaking negative monthly growth rate in currency circulation.

The Dominance of Physical Cash: Despite the introduction of the eNaira (CBDC), physical currency circulation rebounded aggressively in 2024, reaching an all-time high of ₦4.14 Trillion. This suggests that physical cash remains the bedrock of Nigerian domestic commerce.

International Outflow Intensity: By calculating the ratio between international payments and local cash, we observed that Nigeria's external debt service and remittances place significant demand on the country's monetary base, highlighting the ongoing pressure on foreign exchange reserves.
